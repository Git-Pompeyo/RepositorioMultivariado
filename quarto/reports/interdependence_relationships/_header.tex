% Remember the current part header text
\newcommand{\CurrentPartHeader}{}

% Patch \part to store the title
\let\OldPart\part
\renewcommand{\part}[1]{%
  \gdef\CurrentPartHeader{\thepart.~#1}%
  \OldPart{#1}%
}

% Chapter -> leftmark (for odd pages)
\renewcommand{\chaptermark}[1]{%
  \markboth{\thechapter.~#1}{}%
}

% Stop sections/subsections from overwriting marks
\renewcommand{\sectionmark}[1]{}
\renewcommand{\subsectionmark}[1]{}
